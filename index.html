// FUNGSI CEK STATUS NUNGGAK/LUNAS WARGA DI DASBOR (DIPERBARUI) async function cekStatusWarga(source = 'auto') { const container = document.getElementById('beranda-status-iuran'); // Sinkronisasi tahun antara tab Iuran dan Profil agar tidak bentrok const dropIuran = document.getElementById('input-tahun-iuran'); const dropProfil = document.getElementById('profil-rekap-tahun'); let tahunDipilih = new Date().getFullYear(); // AMBIL JENIS IURAN DARI DROPDOWN const dropJenis = document.getElementById('input-jenis-iuran'); let jenisDipilih = dropJenis ? dropJenis.value : 'IuranPaguyuban'; if (source === 'iuran' && dropIuran) { tahunDipilih = dropIuran.value; if (dropProfil) dropProfil.value = tahunDipilih; } else if (source === 'profil' && dropProfil) { tahunDipilih = dropProfil.value; if (dropIuran) dropIuran.value = tahunDipilih; } else { tahunDipilih = dropIuran ? dropIuran.value : new Date().getFullYear(); } if(container) { container.classList.add('shimmer'); container.innerHTML = ''; } // Tampilkan loading di UI Bulan Iuran saat mengecek status tahun baru const gridIuran = document.getElementById('grid-bulan'); if (gridIuran) { const els = gridIuran.querySelectorAll('input[type="checkbox"]'); els.forEach(el => { el.checked = false; el.disabled = true; el.nextElementSibling.innerHTML = ''; }); } const loadingInd = document.getElementById('loading-tahun-indicator'); if(loadingInd) loadingInd.style.display = 'block'; // MENGIRIM TAHUN DAN JENIS IURAN KE BACKEND let res = await fetchAPI("cekStatusBulanIni", { username: getUserUsername(), tahun: tahunDipilih, jenisIuran: jenisDipilih }); if(container) container.classList.remove('shimmer'); if(loadingInd) loadingInd.style.display = 'none'; if(res && res.data) { globalIuranLunasWarga = res.data.lunasDetail || []; let labelKas = jenisDipilih === 'IuranBlok9' ? 'Iuran Blok 9' : 'Iuran Kas RT'; if(container) { if(res.data.status === 'Lunas') { container.innerHTML = `

Status ${labelKas}

LUNAS

Terima kasih telah membayar tepat waktu.

`; container.className = "bg-green-50/50 p-4 rounded-2xl shadow-sm border border-green-200 mb-6 flex items-center justify-between"; } else { container.innerHTML = `

Status ${labelKas}

NUNGGAK

Tunggakan: ${res.data.bulan}

`; container.className = "bg-red-50 p-4 rounded-2xl shadow-sm border border-red-200 mb-6 flex items-center justify-between"; } } const alertNunggak = document.getElementById('iuran-alert-nunggak'); const alertLunas = document.getElementById('iuran-alert-lunas'); const textNunggak = document.getElementById('iuran-text-nunggak'); if (res.data.status === 'Lunas') { if (alertNunggak) alertNunggak.classList.add('hidden'); if (alertLunas) alertLunas.classList.remove('hidden'); } else { if (alertLunas) alertLunas.classList.add('hidden'); if (alertNunggak) alertNunggak.classList.remove('hidden'); if (textNunggak) textNunggak.innerText = res.data.bulan; } renderOpsiPembayaranBulanIuran(globalIuranLunasWarga); renderRekapProfil(res.data.bulan, globalIuranLunasWarga, tahunDipilih); hitungTotalIuran(); } else { if(container) container.innerHTML = `

Gagal memuat status.

`; renderRekapProfil("ERROR"); } } // FUNGSI TRANSPARANSI KAS / KEUANGAN (BERANDA)

// --- DASHBOARD ADMIN STATS --- async function loadAdminDashboardStats() { const elIuran = document.getElementById('count-iuran-menunggu'); const elSuratLapor = document.getElementById('count-surat-laporan'); const elLunas = document.getElementById('count-warga-lunas'); const elNunggak = document.getElementById('count-warga-nunggak'); if(elIuran) elIuran.innerHTML = ''; if(elSuratLapor) elSuratLapor.innerHTML = ''; if(elLunas) elLunas.innerHTML = ''; if(elNunggak) elNunggak.innerHTML = ''; try { const [resIuran, resSurat, resLaporan, resWarga] = await Promise.all([ fetchAPI("getIuran"), fetchAPI("getSurat"), fetchAPI("getLaporan"), fetchAPI("getWarga") ]); let pendingIuran = 0; let pendingSuratLaporan = 0; let countLunas = 0; let countNunggak = 0; if(resIuran && resIuran.data) { pendingIuran = resIuran.data.filter(item => item.status.toLowerCase().includes('menunggu')).length; } if(resSurat && resSurat.data) { pendingSuratLaporan += resSurat.data.filter(item => item.status.toLowerCase().includes('menunggu') || item.status.toLowerCase().includes('diproses')).length; } if(resLaporan && resLaporan.data) { pendingSuratLaporan += resLaporan.data.filter(item => item.status.toLowerCase().includes('menunggu') || item.status.toLowerCase().includes('diproses')).length; } // Pastikan data Warga berhasil diambil untuk menghitung Lunas & Nunggak if(resWarga && resWarga.data && resWarga.data.length > 0) { countLunas = resWarga.data.filter(w => w.status_bulan_ini === 'Lunas').length; countNunggak = resWarga.data.filter(w => w.status_bulan_ini === 'Nunggak').length; // Simpan ke global untuk pencarian globalWargaData = resWarga.data; } if(elIuran) elIuran.innerText = pendingIuran; if(elSuratLapor) elSuratLapor.innerText = pendingSuratLaporan; if(elLunas) elLunas.innerText = countLunas; if(elNunggak) elNunggak.innerText = countNunggak; } catch (error) { console.error("Gagal memuat statistik", error); if(elIuran) elIuran.innerText = "-"; if(elSuratLapor) elSuratLapor.innerText = "-"; if(elLunas) elLunas.innerText = "-"; if(elNunggak) elNunggak.innerText = "-"; } }
